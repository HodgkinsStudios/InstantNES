cmake_minimum_required(VERSION 3.16)
project(InstantNES VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Optimized builds without fast-math, unsafe aliasing, or timing shortcuts.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(IN_BUILD_FRONTEND "Build frontend" ON)
add_library(instantnes_core SHARED src/core/instantnes.c src/core/mapper.c src/core/cpu.c src/core/ppu.c src/core/apu.c)
target_include_directories(instantnes_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(instantnes_core PRIVATE IN_BUILD_SHARED)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(instantnes_core PRIVATE -fno-strict-aliasing)
endif()
set_target_properties(instantnes_core PROPERTIES C_VISIBILITY_PRESET hidden OUTPUT_NAME instantnes VERSION 0.1 SOVERSION 0)
if(IN_BUILD_FRONTEND)
  find_package(X11 REQUIRED)
  add_executable(instantnes src/frontend/x11_main.c)
  target_link_libraries(instantnes PRIVATE instantnes_core X11::X11 m ${CMAKE_DL_LIBS})
  set_target_properties(instantnes_core PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
include(GNUInstallDirs)
install(TARGETS instantnes_core LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/instantnes.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
